function uniq(list) {
  var seen = {}, out = [];
  (list || []).forEach(function (x) {
    var k = String(x || '').replace(/\s+/g, ' ').trim();
    if (!k) return;
    var low = k.toLowerCase();
    if (seen[low]) return;
    seen[low] = true;
    out.push(k);
  });
  return out;
}
function headerLine(text, keys) {
  var re = new RegExp('^\\s*(?:' + keys + ')\\s*[:\\-]\\s*(.+)$', 'i');
  var lines = String(text).split(/\r?\n/);
  for (var i = 0; i < lines.length; i++) {
    var m = lines[i].match(re);
    if (m && m[1].trim()) return m[1].trim();
  }
  return '';
}
function sentences(text) {
  var parts = String(text).replace(/\r/g, '').split(/\n+/);
  var out = [];
  parts.forEach(function (p) {
    p = p.replace(/^[-*•]+\s*/, '').replace(/^\d+[.)]\s*/, '').trim();
    if (p.length >= 3) out.push(p);
  });
  return out;
}
function speakerSplit(line) {
  var m = String(line).match(/^([A-Z][a-zA-Z.'-]{1,20}(?:\s+[A-Z][a-zA-Z.'-]{1,20})?)\s*[:\-]\s+(.+)$/);
  if (m) return { speaker: m[1].trim(), text: m[2].trim() };
  return { speaker: '', text: String(line) };
}
function extractPeople(text) {
  var people = [];
  var hdr = headerLine(text, 'attendees|participants|present|people|team');
  if (hdr) hdr.split(/[,;/]| and /i).forEach(function (n) {
    n = n.trim();
    if (n.length > 1) people.push(n);
  });
  String(text).split(/\n/).forEach(function (line) {
    var sp = speakerSplit(line.trim());
    if (sp.speaker) people.push(sp.speaker);
  });
  return uniq(people);
}
function isQuestion(s) {
  return /\?/.test(s) || /\b(should we|do we|can we|what if|open question|undecided|not sure|parking lot)\b/i.test(s);
}
function isDecision(s) {
  return /\b(decid(e|ed|ing)|decision|agreed|approve|approval|proposal|going with|we will go|adopt|ship|depriorit|confirmed|finalized)\b/i.test(s);
}
function isAction(s) {
  return /\b(i('ll| will)|we('ll| will)|i can|we can|can you|please |need to|needs to|has to|should |must |todo|action|follow[- ]up|own the|handle the|targeting|assign|due |deadline|by (eod|monday|tuesday|wednesday|thursday|friday|end))\b/i.test(s);
}
function isNext(s) {
  return /\b(next meeting|next sync|next week|follow[- ]up|tomorrow|eod|same time|schedule|reconvene)\b/i.test(s);
}
function ownerOf(speaker, s) {
  var m = s.match(/\b([A-Z][a-z]{2,15})\s+(will|can own|owns|is handling|to handle)\b/);
  if (m) return m[1];
  if (speaker && /^(i|'ll|i'll|i will|i can)\b/i.test(s)) return speaker;
  if (speaker) return speaker;
  return 'Unassigned';
}
function deadlineOf(s) {
  var m = s.match(/\b(?:by\s+)?((?:next\s+)?(?:monday|tuesday|wednesday|thursday|friday|week|month)|eod|end of (?:day|week|month)|\d{4}-\d{2}-\d{2})\b/i);
  return m ? m[0].replace(/^by\s+/i,'') : '—';
}
function summarizeNow() {
  var raw = document.getElementById('notes').value.replace(/\r\n/g, '\n').trim();
  if (!raw) { alert('Paste meeting notes first.'); return; }
  var title = document.getElementById('title').value.trim() || headerLine(raw, 'meeting|title|subject') || 'Untitled Meeting';
  var date = document.getElementById('date').value || headerLine(raw, 'date') || '[Not specified]';
  var organizer = headerLine(raw, 'organizer|chair|hosted by|host');
  var people = extractPeople(raw);
  var actions = [], decisions = [], questions = [], nexts = [], points = [];
  raw.split(/\n/).forEach(function (line) {
    line = line.trim();
    if (!line) return;
    if (/^(meeting|title|date|attendees|participants|present|organizer|location|time)\s*[:\-]/i.test(line)) return;
    var sp = speakerSplit(line);
    var s = sp.text;
    if (isQuestion(s)) questions.push(s);
    else if (isDecision(s)) decisions.push(s);
    else if (isAction(s)) actions.push({ text: s, owner: ownerOf(sp.speaker, s), deadline: deadlineOf(s) });
    else if (isNext(s)) nexts.push((sp.speaker ? sp.speaker + ': ' : '') + s);
    else if (s.length >= 12) points.push((sp.speaker ? sp.speaker + ' — ' : '') + s);
  });
  if (!actions.length) {
    sentences(raw).forEach(function (s) {
      if (/^(meeting|title|date|attendees)\s*[:\-]/i.test(s)) return;
      if (s.length >= 12 && s.length <= 200 && !isQuestion(s)) {
        var sp = speakerSplit(s);
        actions.push({ text: sp.text, owner: ownerOf(sp.speaker, sp.text), deadline: deadlineOf(s) });
      }
    });
  }
  if (!nexts.length) {
    sentences(raw).filter(function (s) {
      return s.length > 8 && !/^(meeting|title|date|attendees)\s*[:\-]/i.test(s);
    }).slice(-2).forEach(function (s) { nexts.push(s); });
  }
  if (!points.length) {
    sentences(raw).forEach(function (s) {
      if (s.length >= 8 && !/^(meeting|title|date|attendees)\s*[:\-]/i.test(s)) points.push(s);
    });
  }
  var seenA = {}, actionUniq = [];
  actions.forEach(function (a) {
    var k = String(a.text).toLowerCase();
    if (seenA[k]) return;
    seenA[k] = true;
    actionUniq.push(a);
  });
  actions = actionUniq.slice(0, 12);
  decisions = uniq(decisions);
  questions = uniq(questions);
  nexts = uniq(nexts);
  points = uniq(points).slice(0, 8);
  var md = '# Meeting Minutes: ' + title + '\n\n';
  md += '**Date:** ' + date + '  \n';
  md += '**Organizer:** ' + (organizer || (people[0] || '[Not specified]')) + '  \n';
  md += '**Attendees:** ' + (people.length ? people.join(', ') : '[Not listed in notes]') + '\n\n';
  md += '## Executive Summary\n';
  md += 'Summary of "' + title + '"' + (date !== '[Not specified]' ? ' on ' + date : '') + '. ';
  if (people.length) md += 'People: ' + people.join(', ') + '. ';
  md += points.length + ' discussion point(s), ' + decisions.length + ' decision(s), ' + actions.length + ' action(s), ' + questions.length + ' question(s).\n\n';
  md += '## Key Discussion Points\n';
  md += points.length ? points.map(function (p) { return '- ' + p; }).join('\n') : '- ' + raw.split(/\n/).filter(Boolean)[0];
  md += '\n\n## Decisions\n';
  md += decisions.length ? decisions.map(function (d, i) { return (i + 1) + '. ' + d; }).join('\n') : 'No explicit decision words were found. Review discussion points.';
  md += '\n\n## Action Items\n\n| # | Action Item | Owner | Deadline |\n|---|-------------|-------|----------|\n';
  if (actions.length) {
    actions.forEach(function (a, i) {
      md += '| ' + (i + 1) + ' | ' + String(a.text).replace(/\|/g, '/') + ' | ' + a.owner + ' | ' + a.deadline + ' |\n';
    });
  } else {
    md += '| 1 | Follow up on the topics in these notes | Unassigned | — |\n';
  }
  md += '\n## Open Questions\n';
  md += questions.length ? questions.map(function (q) { return '- ' + q; }).join('\n') : '- None found (no question marks in the notes).';
  md += '\n\n## Next Steps\n';
  md += nexts.length ? nexts.map(function (n) { return '- ' + n; }).join('\n') : '- Review the discussion points and assign owners.';
  md += '\n';
  document.getElementById('out').value = md;
  var st = document.getElementById('status');
  st.style.display = 'block';
  st.textContent = people.length + ' attendee(s) · ' + actions.length + ' action(s) · ' + questions.length + ' question(s) · ' + nexts.length + ' next step(s)';
}
function copyMinutes() {
  var box = document.getElementById('out');
  if (!box.value) { alert('Summarize first.'); return; }
  navigator.clipboard.writeText(box.value);
  var st = document.getElementById('status');
  st.style.display = 'block';
  st.textContent = 'Copied.';
}
function clearAll() {
  document.getElementById('notes').value = '';
  document.getElementById('title').value = '';
  document.getElementById('date').value = '';
  document.getElementById('out').value = '';
  document.getElementById('status').style.display = 'none';
}
